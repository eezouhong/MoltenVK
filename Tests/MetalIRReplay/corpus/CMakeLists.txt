cmake_minimum_required(VERSION 3.20)
project(MetalIRCorpus LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "The ABI-9 compiler helper and provider guard require macOS")
endif()
set(SPIRV_CROSS_INCLUDE "" CACHE PATH "Pinned SPIRV-Cross headers")
set(SPIRV_CROSS_LIBDIR "" CACHE PATH "Pinned MoltenVK SPIRV-Cross static libraries")
set(SPIRV_CROSS_NAMESPACE "MVK_spirv_cross" CACHE STRING "Namespace used when building the static libraries")
if(NOT EXISTS "${SPIRV_CROSS_INCLUDE}/spirv_msl.hpp")
    message(FATAL_ERROR "Set SPIRV_CROSS_INCLUDE")
endif()
add_library(no_metal_guard SHARED no_metal_guard.cpp)
add_executable(compile_module compile_module.cpp)
target_compile_features(no_metal_guard PRIVATE cxx_std_17)
target_compile_features(compile_module PRIVATE cxx_std_17)
target_compile_definitions(compile_module PRIVATE SPIRV_CROSS_NAMESPACE_OVERRIDE=${SPIRV_CROSS_NAMESPACE})
target_include_directories(compile_module PRIVATE "${SPIRV_CROSS_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../MoltenVK/MoltenVK/GPUObjects")
foreach(LIB msl glsl core)
    set(ARCHIVE "${SPIRV_CROSS_LIBDIR}/libspirv-cross-${LIB}.a")
    if(NOT EXISTS "${ARCHIVE}")
        message(FATAL_ERROR "Missing pinned archive: ${ARCHIVE}")
    endif()
    target_link_libraries(compile_module PRIVATE "${ARCHIVE}")
endforeach()
